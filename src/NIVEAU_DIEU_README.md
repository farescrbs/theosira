### 1. Algorithme Génétique

```typescript
// Pseudo-code simplifié

function evolveGeneration() {
  // 1. Évaluation
  strategies.forEach(s => s.fitness = evaluateFitness(s));
  
  // 2. Sélection élite (20%)
  elite = strategies.sort(by fitness).slice(0, 20%);
  
  // 3. Reproduction
  while (newGeneration.length < 50) {
    parent1 = tournamentSelection(elite, 3);
    parent2 = tournamentSelection(elite, 3);
    
    child = crossover(parent1, parent2);
    
    if (random() < 0.3) {
      child = mutate(child);
    }
    
    newGeneration.push(child);
  }
  
  // 4. Remplacement
  strategies = newGeneration;
  generation++;
}

// Exécution: setInterval(evolveGeneration, 60000);
```

### 2. Réseau de Neurones

```typescript
// Architecture: 20 → 15 → 10 → 5

function neuralNetwork(input: number[]) {
  let layer = input; // 20 inputs
  
  // Hidden layers
  layer = activate(matmul(layer, weights[0])); // → 15
  layer = activate(matmul(layer, weights[1])); // → 10
  layer = activate(matmul(layer, weights[2])); // → 5
  
  return layer; // Outputs: predictions
}

function activate(x: number[]) {
  return x.map(v => 1 / (1 + Math.exp(-v))); // Sigmoid
}
```

### 3. Q-Learning (Reinforcement)

```typescript
// Apprentissage par renforcement

function updateQValue(state, action, reward, nextState) {
  const alpha = 0.1; // Learning rate
  const gamma = 0.9; // Discount factor
  
  const currentQ = qValues.get(state, action);
  const maxNextQ = max(qValues.get(nextState, *));
  
  const newQ = currentQ + alpha * (
    reward + gamma * maxNextQ - currentQ
  );
  
  qValues.set(state, action, newQ);
}
```
